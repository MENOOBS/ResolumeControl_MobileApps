class ResolumeControlScreen extends StatefulWidget {
  @override
  _ResolumeControlScreenState createState() => _ResolumeControlScreenState();
}

class _ResolumeControlScreenState extends State<ResolumeControlScreen> {
  String resolumeIP = "192.168.100.9";
  final int resolumePort = 8080;
  List<String> imageUrls = [];
  List<dynamic> _decks = [];
  int? _selectedDeckId;
  TextEditingController ipController = TextEditingController();
  int? selectedIndex;

  @override
  void initState() {
    super.initState();
    _loadDecks(); // Load decks when the screen initializes
    loadSavedIP();
    fetchClipData();
    _checkGuideStatus();
  }

  Future<void> _checkGuideStatus() async {
    SharedPreferences prefs = await SharedPreferences.getInstance();
    bool hasSeenGuide = prefs.getBool('hasSeenGuide') ?? false;
    if (!hasSeenGuide) {
      _showGuideDialog();
    }
  }

  Future<void> _setGuideStatus(bool value) async {
    SharedPreferences prefs = await SharedPreferences.getInstance();
    await prefs.setBool('hasSeenGuide', value);
  }

  void _showGuideDialog() {
    bool dontShowAgain = false;

    showDialog(
      context: context,
      builder: (context) {
        return StatefulBuilder(
          builder: (context, setState) {
            return AlertDialog(
              title: Text("Panduan Menggunakan Aplikasi Ini"),
              content: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  Text(
                    "1. Masukkan IP Address : 192.168.18.43 ke dalam aplikasi.\n"
                    "2. Tekan tombol Simpan.\n"
                    "3. Pastikan  perangkat berada dalam WIFI POV.\n"
                    "4. Jika Konten tidak muncul, periksa koneksi dan coba lagi.",
                  ),
                  Row(),
                ],
              ),
              actions: [
                TextButton(
                  onPressed: () {
                    Navigator.of(context).pop();
                  },
                  child: Text("OK"),
                ),
              ],
            );
          },
        );
      },
    );
  }

  Future<void> loadSavedIP() async {
    SharedPreferences prefs = await SharedPreferences.getInstance();
    String? savedIP = prefs.getString("resolumeIP");
    if (savedIP != null) {
      setState(() {
        resolumeIP = savedIP;
        ipController.text = savedIP;
      });
      fetchClipData();
    }
  }

  // Function to load decks from API
  Future<void> _loadDecks() async {
    print("Starting to load decks...");
    try {
      final response = await http.get(
        Uri.parse('http://$resolumeIP:$resolumePort/api/v1/composition'),
        headers: {'Content-Type': 'application/json'},
      );

      print("API Response status: ${response.statusCode}");
      print(
        "API Response body: ${response.body.substring(0, min(100, response.body.length))}...",
      ); // Print first 100 chars

      if (response.statusCode == 200) {
        final jsonResponse = jsonDecode(response.body);
        print(
          "Parsed JSON contains decks: ${jsonResponse.containsKey('decks')}",
        );

        if (jsonResponse.containsKey('decks')) {
          print("Number of decks found: ${jsonResponse['decks'].length}");

          setState(() {
            _decks = jsonResponse["decks"] ?? [];
            print("Updated _decks list. New length: ${_decks.length}");
            // Clear selection when reloading
            if (_decks.isNotEmpty) {
              _selectedDeckId = _decks.first["id"]; // Set default ke deck pertama
              _selectDeck(_selectedDeckId!); // Pilih deck pertama secara otomatis
            }
          });
        } else {
          print("WARNING: 'decks' key not found in JSON response");
          setState(() {
            _decks = [];
          });
        }
      } else {
        // Handle error
        print('Failed to load decks: ${response.statusCode}');
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Failed to load decks. Please try again.')),
        );
      }
    } catch (e) {
      print('Error loading decks: $e');
      ScaffoldMessenger.of(
        context,
      ).showSnackBar(SnackBar(content: Text('Error loading decks: $e')));
    }
  }

  // Function to select a deck

  Future<void> _selectDeck(int deckId) async {
    try {
      final response = await http.post(
        Uri.parse(
          'http://$resolumeIP:$resolumePort/api/v1/composition/decks/by-id/$deckId/select',
        ),
        headers: {'Content-Type': 'application/json'},
      );

      if (response.statusCode == 200) {
        // Update the selected status in the local deck list
        setState(() {
          for (var deck in _decks) {
            // Set selected to false for all decks
            deck["selected"]["value"] = deck["id"] == deckId;
          }
        });

        // Show success message or update UI as needed
        // ScaffoldMessenger.of(
        //   context,
        // ).showSnackBar(SnackBar(content: Text('Deck selected successfully')));
      } else {
        // Handle error
        print('Failed to select deck: ${response.statusCode}');
        // ScaffoldMessenger.of(
        //   context,
        // ).showSnackBar(SnackBar(content: Text('Failed to select deck')));
      }
    } catch (e) {
      print('Error: $e');
      // ScaffoldMessenger.of(
      //   context,
      // ).showSnackBar(SnackBar(content: Text('Error: $e')));
    }
  }

  Future<void> saveIP(String ip) async {
    SharedPreferences prefs = await SharedPreferences.getInstance();
    await prefs.setString("resolumeIP", ip);
    setState(() {
      resolumeIP = ip;
    });
    fetchClipData();
  }

  Future<void> fetchClipData() async {
    List<String> urls = [];
    int layerIndex = 1;

    try {
      final response = await http.get(
        Uri.parse(
          "http://$resolumeIP:$resolumePort/api/v1/composition/layers/1",
        ),
      );

      if (response.statusCode == 200) {
        var decodedResponse = jsonDecode(response.body);
        var clips = decodedResponse['clips'];

        if (clips is List) {
          for (int clipIndex = 0; clipIndex < clips.length; clipIndex++) {
            String thumbnailUrl =
                "http://$resolumeIP:$resolumePort/api/v1/composition/layers/$layerIndex/clips/${clipIndex + 1}/thumbnail";
            urls.add(thumbnailUrl);
          }
        }

        setState(() {
          imageUrls = urls;
        });
      }
    } catch (e) {
      print("❌ Error mengambil data Resolume: $e");
    }
  }

  Future<void> selectClip(int layer, int clip) async {
    setState(() {
      selectedIndex = clip - 1;
    });

    try {
      final response = await http.post(
        Uri.parse(
          "http://$resolumeIP:$resolumePort/api/v1/composition/layers/$layer/clips/${clip}/connect",
        ),
      );

      if (response.statusCode == 200 || response.statusCode == 204) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(
            content: Text('Berhasil dipilih!'),
            backgroundColor: Colors.green,
            behavior: SnackBarBehavior.floating,
          ),
        );
      }
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text('Gagal memilih clip: $e'),
          backgroundColor: Colors.red,
          behavior: SnackBarBehavior.floating,
        ),
      );
    }

    // Reset selection after animation
    Future.delayed(Duration(milliseconds: 500), () {
      if (mounted) {
        setState(() {
          selectedIndex = null;
        });
      }
    });
  }

                          8,

                        child: DropdownButton<int>(
                          isExpanded: true,
                          dropdownColor:
                              Colors.black26, // Warna background dropdown list
                          hint: Text(
                            "Pilih Deck",
                            overflow: TextOverflow.ellipsis,
                            style: TextStyle(
                              color: Colors.white, // Warna teks dropdown
                              fontSize: 16,
                              fontWeight: FontWeight.bold,
                            ),
                          ),
                          value: _selectedDeckId,

                            Icons.arrow_drop_down,
                            color: Colors.white,
                          ), // Warna ikon dropdown

                            setState(() {
                              _selectedDeckId = newValue;
                            });
                            // Call function to handle deck selection
                            if (newValue != null) {
                              _selectDeck(newValue);
                            }
                          },

                              _decks.map<DropdownMenuItem<int>>((deck) {
                                return DropdownMenuItem<int>(
                                  value: deck["id"],

                                      color: Colors.white,
                                    ), // Warna teks dropdown item
                                  ),

                        ),

                    ),

            icon: Icon(Icons.refresh),

            },

          IconButton(
            icon: Icon(Icons.help_outline),

                      margin: EdgeInsets.symmetric(horizontal: 16),
